# Code Agent TUI Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 code_agent 添加终端交互式 TUI 模式，用户可以直接在终端中与 Agent 对话，实时观察 ReAct 推理过程（thinking → tool_call → tool_result → message），无需启动 HTTP 服务器。

**Architecture:** TUI 作为独立的 `cmd/tui/main.go` 入口，复用 `internal/orchestrator` 的 `ProcessMessageStreamFull` 方法获取事件流，通过 Bubble Tea 框架渲染。TUI 直接连接到本地 Agent 后端（in-process 模式）或通过 HTTP SSE 连接到远程 Agent 服务器（remote 模式）。默认 in-process 模式避免启动 HTTP 服务。

**Tech Stack:**
- [Bubble Tea](https://github.com/charmbracelet/bubbletea) - TUI 框架（Elm 架构）
- [Lip Gloss](https://github.com/charmbracelet/lipgloss) - 终端样式
- [Glamour](https://github.com/charmbracelet/glamour) - Markdown 终端渲染
- [Bubble](https://github.com/charmbracelet/bubbles) - 预构建组件（textarea, viewport, spinner）

## 设计决策

### 1. In-Process vs Remote 模式

| 模式 | 描述 | 适用场景 |
|------|------|---------|
| `--mode=local` (默认) | TUI 进程内直接创建 Orchestrator，无 HTTP | 本地开发，单用户 |
| `--mode=remote --addr=host:port` | 通过 SSE 连接远程 Agent 服务 | 连接已运行的 Docker 服务 |

### 2. 消息渲染策略

- **thinking 事件**: 灰色斜体，可折叠
- **tool_call 事件**: 蓝色加粗工具名 + 折叠的 JSON args
- **tool_result 事件**: 绿色/红色（成功/失败），内容截断显示
- **message 事件**: Glamour 渲染 Markdown
- **error 事件**: 红色背景高亮
- **step_start 事件**: 分隔符 + step 编号

### 3. 会话管理

- 启动时自动创建新 session（或 `--session=<id>` 恢复）
- `/new` 命令创建新会话
- `/sessions` 命令列出会话
- `/quit` 或 Ctrl+C 退出

```go
	// ListSessions returns a list of existing session summaries.
	ListSessions(ctx context.Context) ([]SessionSummary, error)

	// Close releases any resources held by the backend.
	Close() error
}

// SessionSummary is a minimal view of a session for TUI display.

	ID             string
	MessageCount int
	LastMessage    string
	LastUpdateTime string
}
```

```go
package main

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"

	"github.com/agent/code_agent/internal/models"
	"github.com/agent/code_agent/internal/tui"
)

// RemoteBackend connects to a running code-agent HTTP server via SSE.
type RemoteBackend struct {
	baseURL    string
	httpClient *http.Client
}

func NewRemoteBackend(addr string) *RemoteBackend {
	return &RemoteBackend{
		baseURL:    fmt.Sprintf("http://%s", addr),
		httpClient: &http.Client{},
	}
}

func (r *RemoteBackend) SendMessage(ctx context.Context, sessionID, message string) (<-chan models.ReactStreamEvent, error) {
	eventCh := make(chan models.ReactStreamEvent, 64)

	body := fmt.Sprintf(`{"session_id":"%s","message":%s}`, sessionID, jsonStr(message))
	req, err := http.NewRequestWithContext(ctx, "POST", r.baseURL+"/api/v1/chat/react-stream", strings.NewReader(body))
	if err != nil {
		return nil, err
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Accept", "text/event-stream")

	resp, err := r.httpClient.Do(req)
	if err != nil {
		return nil, fmt.Errorf("failed to connect to agent: %w", err)
	}
	if resp.StatusCode != http.StatusOK {
		resp.Body.Close()
		return nil, fmt.Errorf("agent returned status %d", resp.StatusCode)
	}

	go r.readSSE(ctx, resp.Body, eventCh)
	return eventCh, nil
}

func (r *RemoteBackend) readSSE(ctx context.Context, body io.ReadCloser, ch chan<- models.ReactStreamEvent) {
	defer close(ch)
	defer body.Close()

	scanner := bufio.NewScanner(body)
	for scanner.Scan() {
		line := scanner.Text()
		if !strings.HasPrefix(line, "data: ") {
			continue
		}
		data := strings.TrimPrefix(line, "data: ")
		var ev models.ReactStreamEvent
		if err := json.Unmarshal([]byte(data), &ev); err != nil {
			continue
		}
		select {
		case ch <- ev:
		case <-ctx.Done():
			return
		}
	}
}

func (r *RemoteBackend) CreateSession(ctx context.Context) (string, error) {
	req, err := http.NewRequestWithContext(ctx, "POST", r.baseURL+"/api/v1/sessions", strings.NewReader(`{}`))
	if err != nil {
		return "", err
	}
	req.Header.Set("Content-Type", "application/json")
	resp, err := r.httpClient.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	var result struct {
		ID string `json:"id"`
	}
	json.NewDecoder(resp.Body).Decode(&result)
	return result.ID, nil
}

func (r *RemoteBackend) ListSessions(ctx context.Context) ([]tui.SessionSummary, error) {
	req, err := http.NewRequestWithContext(ctx, "GET", r.baseURL+"/api/v1/sessions", nil)
	if err != nil {
		return nil, err
	}
	resp, err := r.httpClient.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	var result struct {
		Sessions []struct {
			ID           string `json:"id"`
			MessageCount int    `json:"message_count"`
			LastPreview  string `json:"last_message_preview"`
			UpdatedAt    string `json:"updated_at"`
		} `json:"sessions"`
	}
	json.NewDecoder(resp.Body).Decode(&result)

	summaries := make([]tui.SessionSummary, len(result.Sessions))
	for i, s := range result.Sessions {
		summaries[i] = tui.SessionSummary{
			ID:             s.ID,
			MessageCount:   s.MessageCount,
			LastMessage:    s.LastPreview,
			LastUpdateTime: s.UpdatedAt,
		}
	}
	return summaries, nil
}

func (r *RemoteBackend) Close() error { return nil }

func jsonStr(s string) string {
	b, _ := json.Marshal(s)
	return string(b)
}
```

```go
// LocalBackend runs the orchestrator in-process without HTTP server.
type LocalBackend struct {
	orch       *orchestrator.Orchestrator
	sessionMgr *session.Manager
	closer     func() // cleanup function for all deps
}

// NewLocalBackend creates a fully-wired local orchestrator.
// It initializes Redis, LLM client, session manager, and orchestrator
// following the same dependency setup as cmd/agent/main.go but without HTTP.
func NewLocalBackend(configPath string) (*LocalBackend, error) {
	cfg, err := config.Load(configPath)
	if err != nil {
		return nil, fmt.Errorf("load config: %w", err)
	}
	if err := cfg.Validate(); err != nil {
		return nil, fmt.Errorf("validate config: %w", err)
	}

	logger, _ := zap.NewProduction()

	// Redis (required)
	rdb := redis.NewClient(&redis.Options{
		Addr:     cfg.Redis.Addr,
		Password: cfg.Redis.Password,
		DB:       cfg.Redis.DB,
	})
	if err := rdb.Ping(context.Background()).Err(); err != nil {
		return nil, fmt.Errorf("redis connect: %w", err)
	}

	// Session manager
	sessMgr := session.NewManager(rdb, &cfg.Session, logger)

	// LLM client
	llmClient := llm.NewClient(&cfg.LLM, logger)

	// Orchestrator
	orch := orchestrator.New(llmClient, sessMgr, nil, nil, logger)

	closer := func() {
		rdb.Close()
		logger.Sync()
	}

	return &LocalBackend{orch: orch, sessionMgr: sessMgr, closer: closer}, nil
}

func (l *LocalBackend) SendMessage(ctx context.Context, sessionID, message string) (<-chan models.ReactStreamEvent, error) {
	return l.orch.ProcessMessageStreamFull(ctx, sessionID, message)
}

func (l *LocalBackend) CreateSession(ctx context.Context) (string, error) {
	sess, err := l.sessionMgr.Create(ctx, "tui-user", "")
	if err != nil {
		return "", err
	}
	return sess.ID, nil
}

func (l *LocalBackend) ListSessions(ctx context.Context) ([]tui.SessionSummary, error) {
	sessions, err := l.sessionMgr.ListSessions(ctx, "tui-user")
	if err != nil {
		return nil, err
	}
	summaries := make([]tui.SessionSummary, len(sessions))
	for i, s := range sessions {
		summaries[i] = tui.SessionSummary{
			ID:           s.ID,
			MessageCount: s.MessageCount,
			LastMessage:  s.LastMessagePreview,
		}
	}
	return summaries, nil
}

func (l *LocalBackend) Close() error {
	if l.closer != nil {
		l.closer()
	}
	return nil
}
```

```go
package main

import (
	"fmt"
	"strings"

	"github.com/agent/code_agent/internal/models"
	"github.com/charmbracelet/glamour"
)

// markdownRenderer is a shared glamour renderer for assistant messages.
var markdownRenderer *glamour.TermRenderer

func init() {
	var err error
	markdownRenderer, err = glamour.NewTermRenderer(
		glamour.WithAutoStyle(),
		glamour.WithWordWrap(100),
	)
	if err != nil {
		// fallback: no markdown rendering
		markdownRenderer = nil
	}
}

// renderEvent converts a ReactStreamEvent to a styled terminal string.
func renderEvent(ev models.ReactStreamEvent) string {
	switch ev.Type {
	case "step_start":
		return stepDividerStyle.Render(
			fmt.Sprintf("─── Step %d/%d [%s] ───", ev.Step, ev.MaxSteps, ev.Intent),
		)

	case "thinking":
		content := ev.Content
		if len(content) > 200 {
			content = content[:200] + "..."
		}
		return thinkingStyle.Render("💭 " + content)

	case "tool_call":
		args := ev.ToolArgs
		if len(args) > 100 {
			args = args[:100] + "..."
		}
		return toolCallStyle.Render(fmt.Sprintf("🔧 %s(%s)", ev.ToolName, args))

	case "tool_result":
		content := ev.Content
		if len(content) > 300 {
			content = content[:300] + "\n    ... (truncated)"
		}
		if ev.IsError {
			return toolErrorStyle.Render("✗ " + content)
		}
		return toolResultStyle.Render("✓ " + content)

	case "message":
		rendered := ev.Content
		if markdownRenderer != nil {
			if md, err := markdownRenderer.Render(ev.Content); err == nil {
				rendered = strings.TrimSpace(md)
			}
		}
		return assistantMsgStyle.Render(rendered)

	case "rag_context":
		return thinkingStyle.Render("📚 " + ev.Content)

	case "error":
		return errorMsgStyle.Render("❌ Error: " + ev.Content)

	case "done":
		return stepDividerStyle.Render("─── Complete ───")

	default:
		return thinkingStyle.Render(fmt.Sprintf("[%s] %s", ev.Type, ev.Content))
	}
}
```

```go
package main

import (
	"fmt"

	"github.com/charmbracelet/lipgloss"
)

type statusBar struct {
	sessionID string
	step      int
	maxSteps  int
	state     string // "idle", "thinking", "tool_call", "streaming"
	width     int
}

func newStatusBar() statusBar {
	return statusBar{state: "idle"}
}

func (s statusBar) View() string {
	// Left side: session info
	left := fmt.Sprintf(" 📋 %s", truncateID(s.sessionID))

	// Center: current state
	var stateIcon string
	switch s.state {
	case "idle":
		stateIcon = "⏸ Idle"
	case "thinking":
		stateIcon = "🧠 Thinking..."
	case "tool_call":
		stateIcon = "🔧 Executing tool..."
	case "streaming":
		stateIcon = "💬 Responding..."
	}
	center := stateIcon

	// Right side: step progress
	var right string
	if s.maxSteps > 0 {
		right = fmt.Sprintf("Step %d/%d ", s.step, s.maxSteps)
	} else {
		right = "Ready "
	}

	// Layout: fill the width
	leftW := lipgloss.Width(left)
	centerW := lipgloss.Width(center)
	rightW := lipgloss.Width(right)
	gap := s.width - leftW - centerW - rightW
	if gap < 0 {
		gap = 0
	}
	leftGap := gap / 2
	rightGap := gap - leftGap

	bar := left +
		repeatStr(" ", leftGap) +
		center +
		repeatStr(" ", rightGap) +
		right

	return statusBarStyle.Width(s.width).Render(bar)
}

func truncateID(id string) string {
	if len(id) > 8 {
		return id[:8]
	}
	return id
}

func repeatStr(s string, n int) string {
	if n <= 0 {
		return ""
	}
	result := ""
	for i := 0; i < n; i++ {
		result += s
	}
	return result
}
```

```go
package main

import (
	"context"
	"fmt"
	"strings"

	"github.com/agent/code_agent/internal/models"
	"github.com/agent/code_agent/internal/tui"
	"github.com/charmbracelet/bubbles/spinner"

// ─── Messages ───────────────────────────────────────────────────────────────

type streamEventMsg models.ReactStreamEvent
type streamDoneMsg struct{}
type sessionCreatedMsg string
type errMsg struct{ err error }

// ─── Model ──────────────────────────────────────────────────────────────────

type appModel struct {
	backend   tui.Backend
	sessionID string

	// UI components
	textarea  textarea.Model
	viewport  viewport.Model
	spinner   spinner.Model
	statusBar statusBar

	// State
	chatLines []string // rendered lines for viewport
	streaming bool     // whether agent is currently responding
	width     int
	height    int

func newAppModel(backend tui.Backend) appModel {

	ta := textarea.New()

		},

func (m appModel) Init() tea.Cmd {
	return tea.Batch(
		textarea.Blink,
		m.spinner.Tick,
		m.createSession,
	)
}

// ─── Update ────────────────────────────────────────────────────────────────

func (m appModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	var cmds []tea.Cmd

	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width = msg.Width
		m.height = msg.Height
		m.statusBar.width = msg.Width

		headerH := 0
		statusH := 1
		inputH := m.textarea.Height() + 2
		vpH := m.height - headerH - statusH - inputH - 2
		if vpH < 1 {
			vpH = 1
		}

		if !m.ready {
			m.viewport = viewport.New(m.width, vpH)
			m.viewport.SetContent(strings.Join(m.chatLines, "\n"))
			m.ready = true
		} else {
			m.viewport.Width = m.width
			m.viewport.Height = vpH
		}
		m.textarea.SetWidth(m.width - 2)

	case tea.KeyMsg:
		switch {
		case msg.Type == tea.KeyCtrlC || msg.Type == tea.KeyCtrlD:
			if m.cancelFn != nil {
				m.cancelFn()
			}
			return m, tea.Quit

		case msg.Type == tea.KeyEnter && !m.streaming:
			input := strings.TrimSpace(m.textarea.Value())
			if input == "" {
				break
			}
			m.textarea.Reset()

			// Handle slash commands
			if strings.HasPrefix(input, "/") {
				return m.handleCommand(input)
			}

			// Send message
			m.chatLines = append(m.chatLines, userMsgStyle.Render("You: "+input), "")
			m.viewport.SetContent(strings.Join(m.chatLines, "\n"))
			m.viewport.GotoBottom()
			m.streaming = true
			m.statusBar.state = "thinking"
			return m, m.sendMessage(input)
		}

	case streamEventMsg:
		ev := models.ReactStreamEvent(msg)
		rendered := renderEvent(ev)

		// Update status bar based on event type
		switch ev.Type {
		case "step_start":
			m.statusBar.step = ev.Step
			m.statusBar.maxSteps = ev.MaxSteps
			m.statusBar.state = "thinking"
		case "thinking":
			m.statusBar.state = "thinking"
		case "tool_call":
			m.statusBar.state = "tool_call"
		case "message":
			m.statusBar.state = "streaming"
		case "done":
			m.streaming = false
			m.statusBar.state = "idle"
		case "error":
			m.streaming = false
			m.statusBar.state = "idle"
		}

		if rendered != "" {
			m.chatLines = append(m.chatLines, rendered)
			m.viewport.SetContent(strings.Join(m.chatLines, "\n"))
			m.viewport.GotoBottom()
		}

	case streamDoneMsg:
		m.streaming = false
		m.statusBar.state = "idle"
		m.chatLines = append(m.chatLines, "")

	case sessionCreatedMsg:
		m.sessionID = string(msg)
		m.statusBar.sessionID = string(msg)

	case errMsg:
		m.streaming = false
		m.statusBar.state = "idle"
		m.chatLines = append(m.chatLines, errorMsgStyle.Render("Error: "+msg.err.Error()), "")
		m.viewport.SetContent(strings.Join(m.chatLines, "\n"))
		m.viewport.GotoBottom()

	case spinner.TickMsg:
		var cmd tea.Cmd
		m.spinner, cmd = m.spinner.Update(msg)
		cmds = append(cmds, cmd)
	}

	// Update sub-components
	if !m.streaming {
		var cmd tea.Cmd
		m.textarea, cmd = m.textarea.Update(msg)
		cmds = append(cmds, cmd)
	}

	var cmd tea.Cmd
	m.viewport, cmd = m.viewport.Update(msg)
	cmds = append(cmds, cmd)

	return m, tea.Batch(cmds...)
}

// ─── View ───────────────────────────────────────────────────────────────────

func (m appModel) View() string {
	if !m.ready {
		return "Initializing..."
	}

	var sections []string

	// Chat viewport
	sections = append(sections, m.viewport.View())

	// Status bar
	sections = append(sections, m.statusBar.View())

	// Input area
	if m.streaming {
		sections = append(sections, m.spinner.View()+" Agent is working...")
	} else {
		sections = append(sections, m.textarea.View())
	}

	return lipgloss.JoinVertical(lipgloss.Left, sections...)
}
```

```go
package main

import (
	"flag"
	"fmt"
	"os"

	"github.com/agent/code_agent/internal/tui"
	tea "github.com/charmbracelet/bubbletea"
)

func main() {
	mode := flag.String("mode", "remote", "Backend mode: local (in-process) or remote (SSE)")
	addr := flag.String("addr", "localhost:18080", "Remote agent address (used with --mode=remote)")
	configPath := flag.String("config", "", "Config file path (used with --mode=local)")
	sessionID := flag.String("session", "", "Resume existing session by ID")
	flag.Parse()

	var backend tui.Backend
	var err error

	switch *mode {
	case "remote":
		backend = NewRemoteBackend(*addr)
	case "local":
		backend, err = NewLocalBackend(*configPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "Failed to initialize local backend: %v\n", err)
			os.Exit(1)
		}
	default:
		fmt.Fprintf(os.Stderr, "Unknown mode: %s (use 'local' or 'remote')\n", *mode)
		os.Exit(1)
	}
	defer backend.Close()

	model := newAppModel(backend)
	if *sessionID != "" {
		model.sessionID = *sessionID
		model.statusBar.sessionID = *sessionID
	}

	p := tea.NewProgram(model, tea.WithAltScreen())
	if _, err := p.Run(); err != nil {
		fmt.Fprintf(os.Stderr, "TUI error: %v\n", err)
		os.Exit(1)
	}
}
```

Bubble Tea 的正确做法是在 `Init()` 或 command 中返回 `tea.Cmd`，而不是在 goroutine 中直接推送。需要使用 `tea.Program.Send()` 方法。改为使用 channel + waitForEvent 模式：

在 appModel 中添加 `eventCh` 字段：
```go
type appModel struct {
	// ... existing fields ...
	eventCh <-chan models.ReactStreamEvent // current active event stream
	program *tea.Program               // reference for external sends
}
```

修改 sendMessage 使用 `waitForEvent` 递归命令模式：
```go
func waitForEvent(ch <-chan models.ReactStreamEvent) tea.Cmd {
	return func() tea.Msg {
		ev, ok := <-ch
		if !ok {
			return streamDoneMsg{}
		}
		return streamEventMsg(ev)
	}
}

func (m *appModel) startStream(msg string) tea.Cmd {
	return func() tea.Msg {
		ctx := context.Background()
		eventCh, err := m.backend.SendMessage(ctx, m.sessionID, msg)
		if err != nil {
			return errMsg{err}
		}
		return streamStartMsg{ch: eventCh}
	}
}

type streamStartMsg struct {
	ch <-chan models.ReactStreamEvent
}
```

在 Update 中处理 `streamStartMsg` 后返回 `waitForEvent(m.eventCh)`，每收到一个 `streamEventMsg` 后继续返回 `waitForEvent` 直到 `streamDoneMsg`。

```go
func TestRemoteBackend_SendMessage_SSE(t *testing.T) {
	events := []models.ReactStreamEvent{
		{Type: "step_start", Step: 1, MaxSteps: 10},
		{Type: "thinking", Content: "analyzing..."},
		{Type: "message", Content: "Hello!"},
		{Type: "done"},
	}

	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/event-stream")
		flusher := w.(http.Flusher)
		for _, ev := range events {
			data, _ := json.Marshal(ev)

			flusher.Flush()
		}

	addr := server.URL[7:]
	backend := NewRemoteBackend(addr)

	ch, err := backend.SendMessage(context.Background(), "sess-1", "hello")
	require.NoError(t, err)

	var received []models.ReactStreamEvent
	for ev := range ch {
		received = append(received, ev)
	}

	assert.Len(t, received, 4)
	assert.Equal(t, "step_start", received[0].Type)
```
